import {writeFileSync} from 'node:fs';
const jobs=Array.from({length:8},(_,i)=>Bun.spawn([process.execPath,'balance-runs/cycle-thin-worker.ts',String(Math.floor(i/2)),String(i%2===1)],{stdout:'pipe',stderr:'inherit'}));
const rows=await Promise.all(jobs.map(async p=>{const text=await new Response(p.stdout).text();if(await p.exited)throw Error('Probe failed');return text;}));
writeFileSync('balance-runs/cycle-thin-probe.jsonl',rows.join(''));
